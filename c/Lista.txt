Algoritmo "Exercicio1"
Var
   n1, n2: inteiro
Inicio
   Escreval("Digite dois numeros inteiros:")
   Leia(n1, n2)
   Escreval("Soma: ", n1 + n2)
   Escreval("Subtracao: ", n1 - n2)
   Escreval("Multiplicacao: ", n1 * n2)
   Se n2 <> 0 Entao
      Escreval("Divisao: ", n1 / n2)
   Senao
      Escreval("Divisao por zero")
   FimSe
Fimalgoritmo

Algoritmo "Exercicio6"
Var
   n1, n2, resultado: real
   op: caractere
Inicio
   Escreval("Digite o primeiro numero:")
   Leia(n1)
   Escreval("Digite o segundo numero:")
   Leia(n2)
   Escreval("Digite a operacao (+, -, *, /):")
   Leia(op)
   
   Escolha op
      Caso "+"
         resultado <- n1 + n2
         Escreval("Resultado: ", resultado)
      Caso "-"
         resultado <- n1 - n2
         Escreval("Resultado: ", resultado)
      Caso "*"
         resultado <- n1 * n2
         Escreval("Resultado: ", resultado)
      Caso "/"
         Se n2 <> 0 Entao
            resultado <- n1 / n2
            Escreval("Resultado: ", resultado)
         Senao
            Escreval("Erro: divisao por zero.")
         FimSe
      OutroCaso
         Escreval("Operacao invalida.")
   FimEscolha
Fimalgoritmo
